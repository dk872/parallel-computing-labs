package org.example;

/**
 * ПЗВПКС
 * Лабораторна робота ЛР2.
 * Циклічний запуск 5 разів.
 */
public class Lab2 {
    public static void main(String[] args) {
        System.out.println("Lab2 started");

        // Змінна для накопичення загального часу (якщо потрібно середнє)
        long totalDuration = 0;
        int iterations = 5;

        try {
            Thread.sleep(20000);
        } catch (InterruptedException e) {
            e.printStackTrace();
        }

        for (int i = 0; i < iterations; i++) {
            System.out.println("\n--- Iteration " + (i + 1) + " ---");

            // 1. ВАЖЛИВО: Скидання спільних ресурсів перед новим запуском!
            // Якщо цього не зробити, AtomicInteger 'a' залишиться з попереднім максимумом,
            // а бар'єри можуть бути в некоректному стані.
            refreshData();

            // 2. Ініціалізація НОВИХ об'єктів потоків для кожної ітерації
            T1 t1 = new T1();
            T2 t2 = new T2();
            T3 t3 = new T3();
            T4 t4 = new T4();

            // Початок вимірювання часу
            long startTime = System.currentTimeMillis();

            // Запуск потоків
            t1.start();
            t2.start();
            t3.start();
            t4.start();

            // Очікування завершення
            try {
                t1.join();
                t2.join();
                t3.join();
                t4.join();
            } catch (InterruptedException e) {
                e.printStackTrace();
            }

            // Кінець вимірювання часу
            long endTime = System.currentTimeMillis();
            long duration = endTime - startTime;
            totalDuration += duration;

            System.out.println("Time execution: " + duration + " ms");
        }

        System.out.println("\n--------------------------------");
        System.out.println("Total time: " + totalDuration + " ms");
        System.out.println("Average time: " + (totalDuration / iterations) + " ms");
        System.out.println("Lab2 finished");
    }

    /**
     * Метод для оновлення стану спільних змінних у класі Data.
     * Це критично необхідно для циклічного запуску.
     */
    private static void refreshData() {
        // Скидаємо глобальний максимум у початкове значення
        Data.a.set(Integer.MIN_VALUE);

        // Оскільки CyclicBarrier в Java перевикористовуваний (reusable) автоматично,
        // якщо всі потоки успішно пройшли його, то його можна не перестворювати.
        // Але якщо ви хочете повної надійності (щоб уникнути BrokenBarrierException при помилках),
        // можна розкоментувати рядки нижче (потрібно змінити модифікатор final у Data):

        /*
        Data.B1 = new java.util.concurrent.CyclicBarrier(4);
        Data.B2 = new java.util.concurrent.CyclicBarrier(4);
        Data.B3 = new java.util.concurrent.CyclicBarrier(4); // Якщо ви його додали

        // Семафори сигналів (S2, S3, S4) повинні бути 0.
        // Якщо потік T3 відпрацював коректно, вони вже 0 (він зробив acquire).
        // Але для гарантії можна перестворити:

}
